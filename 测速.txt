export default {
  async fetch(request) {
    const url = new URL(request.url);
    const path = url.pathname.substring(1);

    // 1. 根路径或无效路径返回环保伪装页
    if (path === "") {
      return new Response(DISGUISE_HTML, {
        headers: { "Content-Type": "text/html;charset=UTF-8" },
      });
    }

    // 2. 原有的 locations 逻辑
    if (path === "locations") {
      return locations_cn(request);
    }

    // 3. 原有的测速逻辑
    const match = path.match(/^(\d+)([a-z]{0,2})$/i);
    if (match) {
      const size = parseInt(match[1], 10);
      const unit = match[2].toLowerCase();
      const multipliers = {
        'k': 1000, 'kb': 1000,
        'm': 1000000, 'mb': 1000000,
        'g': 1000000000, 'gb': 1000000000
      };
      const bytes = size * (multipliers[unit] || 1);

      const targetUrl = `https://speed.cloudflare.com/__down?bytes=${bytes}`;
      const headers = new Headers(request.headers);
      headers.set('referer', 'https://speed.cloudflare.com/');

      return fetch(new Request(targetUrl, {
        method: request.method,
        headers: headers,
        body: request.body
      }));
    }

    // 4. 所有其他路径也返回伪装页，确保安全性
    return new Response(DISGUISE_HTML, {
      headers: { "Content-Type": "text/html;charset=UTF-8" },
    });
  }
};

// ... 保持 locations_cn 函数部分完全不变[cite: 1] ...
async function locations_cn(request) {
  const response = await fetch('https://speed.cloudflare.com/locations', {
    headers: { 'referer': 'https://speed.cloudflare.com/' }
  });
  const locations = await response.json();
  // ... 映射逻辑与之前一致[cite: 1] ...
  return new Response(JSON.stringify(locations, null, 4), {
    headers: { 'Content-Type': 'application/json' }
  });
}
